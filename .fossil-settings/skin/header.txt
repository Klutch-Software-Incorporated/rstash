<div class="mainmenu">
<th1>
  html "<a id''hbbtn' href='$home/sitemap' aria-label='Site Map'>&#9776;</a>"
  builtin_request_js hbmenu.js

  foreach {name url expr class} $mainmenu {
    if {![capexpr $expr]} continue

    # If URL starts with "/" it is internal root-relative -> prefix $home
    if {[string match /* $url]} {
      set url $home$url

    # If URL contains a ":" assume it is an absolute scheme (http:, https:, mailto:, etc.) -> leave alone
    } elseif {[string match *:* $url]} {
      # no-op

    # Otherwise it is a bare relative internal path like "brlist" -> prefix "$home/"
    } else {
      set url $home/$url
    }

    html "<a href='$url' class='$class'>$name</a>\n"
  }
</th1>
